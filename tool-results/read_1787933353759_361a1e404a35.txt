     5→     5→ *
     6→     6→ * Customers embed a single tag:
     7→     7→ *   <script async defer src="https://app/api/widget/<slug>/script"></script>
     8→     8→ *
     9→     9→ * The script:
    10→    10→ *   1. Fetches the tenant's WidgetConfig (theme)
    11→    11→ *   2. Generates a visitorId (stored in localStorage) and identifies the visitor
    12→    12→ *   3. Loads socket.io-client from the same origin
    13→    13→ *   4. Connects to the realtime service via the Caddy gateway (XTransformPort=3003)
    14→    14→ *   5. Renders a full chat UI (message list, input, typing indicator)
    15→    15→ *   6. Sends/receives messages in real time via REST + Socket.IO
    16→    16→ *
    17→    17→ * Fully self-contained (no framework), RTL-aware via the config's defaultDirection.
    18→    18→ */

   316→   316→  function sendMessage(){
   317→   317→    var text = inputEl.value.trim();
   318→   318→    if (!text) return;
   319→   319→    inputEl.value = '';
   320→   320→
   321→   321→    // If no conversation yet, create one by sending the first message
   322→   322→    fetch(MESSAGES_URL, {
   323→   323→      method: 'POST',
   324→   324→      headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + state.token },
   325→   325→      body: JSON.stringify({ text: text }),
   326→   326→    })
   327→   327→    .then(function(r){ return r.json(); })
   328→   328→    .then(function(data){
   329→   329→      if (data.message) {
   330→   330→        state.messages.push(data.message);
   331→   331→        renderMessages();
   332→   332→      }
   333→   333→      if (data.conversationId && data.conversationId !== state.conversationId) {
   334→   334→        state.conversationId = data.conversationId;
   335→   335→        // New conversation just created. The socket was already warmed up
   336→   336→        // during identifyVisitor(). If it's connected, join the conversation
   337→   337→        // room NOW so we don't miss rapid agent replies. If it's still
   338→   338→        // connecting, the 'connect' handler will join the room automatically
   339→   339→        // (it checks state.conversationId on connect).
   340→   340→        if (state.socket && state.connected) {
   341→   341→          state.socket.emit('conversation:join', state.conversationId);
   342→   342→        }
   343→   343→      }
   344→   344→      // Start polling for new messages (fallback if Socket.IO fails through proxy)
   345→   345→      startPolling();
   346→   346→    })
   347→   347→    .catch(function(e){ console.error('[sukhan] send failed', e); });
   348→   348→  }

   350→   350→  // ---- Polling fallback (safety net — see DISABLE_POLLING flag above) ----
   351→   351→  // Primary delivery is Socket.IO. This polling runs only as a resilience
   352→   352→  // fallback in case the realtime connection drops or is blocked by a proxy.
   353→   353→  // Interval is 10s in normal operation — long enough to not be chatty, short
   354→   354→  // enough to recover within a tolerable window if Socket.IO fails silently.
   355→   355→  var pollTimer = null;
   356→   356→  var lastPollCount = 0;
   357→   357→  function startPolling(){
   358→   358→    if (DISABLE_POLLING) return; // test-only bypass
   359→   359→    if (pollTimer) return;
   360→   360→    pollTimer = setInterval(function(){
   361→   361→      if (!state.conversationId || !state.token) return;
   362→   362→      fetch(MESSAGES_URL + '?conversationId=' + state.conversationId, {
   363→   363→        headers: { 'Authorization': 'Bearer ' + state.token },
   364→   364→      })
   365→   365→      .then(function(r){ return r.json(); })
   366→   366→      .then(function(data){
   367→   367→        if (data.messages && data.messages.length !== state.messages.length) {
   368→   368→          // New messages arrived — update the list
   369→   369→          var newMsgs = data.messages.slice(state.messages.length);
   370→   370→          for (var i = 0; i < newMsgs.length; i++) {
   371→   371→            // Don't double-add our own messages (already in state.messages)
   372→   372→            var exists = state.messages.some(function(m){ return m.id === newMsgs[i].id; });
   373→   373→            if (!exists) {
   374→   374→              state.messages.push(newMsgs[i]);
   375→   375→            }
   376→   376→          }
   377→   377→          renderMessages();
   378→   378→          if (!state.open) {
   379→   379→            pulseDot.style.display = 'block';
   380→   380→          }
   381→   381→        }
   382→   382→      })
   383→   383→      .catch(function(){});
   384→   384→    }, 10000); // 10s safety net — Socket.IO is the primary delivery path
   385→   385→  }

   393→   393→  function renderMessages(){
   394→   394→    // Clear body except typing indicator
